-- Harden the mobile identity bridge.
--
-- app_users contains authorization and payroll fields. RLS can restrict rows,
-- but it cannot restrict columns, so authenticated mobile clients must never
-- receive table-level INSERT/UPDATE/DELETE privileges on this table.

create schema if not exists private;
revoke all on schema private from public;
grant usage on schema private to authenticated, service_role;

create or replace function private.resolve_mobile_app_user_id()
returns uuid
language sql
stable
security definer
set search_path = ''
as $$
  select candidate.id
  from auth.users auth_user
  join lateral (
    select app_user.id
    from public.app_users app_user
    where app_user.supabase_auth_uid = auth_user.id
       or (
         auth_user.email_confirmed_at is not null
         and auth_user.email is not null
         and lower(app_user.email) = lower(auth_user.email)
       )
    order by
      (app_user.supabase_auth_uid = auth_user.id) desc,
      app_user.created_at asc
    limit 1
  ) candidate on true
  where auth_user.id = (select auth.uid())
  limit 1;
$$;

revoke execute on function private.resolve_mobile_app_user_id() from public, anon, authenticated;
grant execute on function private.resolve_mobile_app_user_id() to authenticated, service_role;

create or replace function private.resolve_mobile_app_user_role()
returns text
language sql
stable
security definer
set search_path = ''
as $$
  select app_user.role
  from public.app_users app_user
  where app_user.id = private.resolve_mobile_app_user_id()
  limit 1;
$$;

revoke execute on function private.resolve_mobile_app_user_role() from public, anon, authenticated;
grant execute on function private.resolve_mobile_app_user_role() to authenticated, service_role;

-- Public wrappers remain security-invoker RPCs. The privileged lookup is kept
